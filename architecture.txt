/mini-redis
│── /cmd                   # Dossier contenant l’exécutable principal
│   ├── main.go            # Point d’entrée du programme
│── /internal              # Contient le cœur de l'application (non exposé en dehors du projet)
│   ├── server.go          # Gestion du serveur TCP
│   ├── storage.go         # Gestion du stockage clé-valeur en mémoire
│   ├── parser.go          # Analyse et exécution des commandes (`SET`, `GET`, `DEL`)
│   ├── expiration.go      # Gestion de l’expiration des clés
│   ├── persistence.go     # Sauvegarde et chargement des données sur disque
│   ├── security.go        # Gestion de l'authentification et des accès sécurisés
│   ├── concurrency.go     # Gestion de la concurrence et des accès multiples
│── /config                # Configuration et paramètres du serveur
│   ├── config.yaml        # Fichier de configuration (port, timeout, etc.)
│── /pkg                   # Bibliothèques réutilisables (exportables pour d’autres projets)
│   ├── logger.go          # Gestion des logs et messages serveur
│   ├── utils.go           # Fonctions utilitaires communes
│── /tests                 # Contient les tests unitaires et d’intégration
│   ├── server_test.go     # Tests du serveur TCP
│   ├── storage_test.go    # Tests du stockage clé-valeur
│   ├── parser_test.go     # Tests du parsing des commandes
│── /docker                # Contient les fichiers Docker et les scripts de déploiement
│   ├── Dockerfile         # Fichier Docker pour exécuter Mini-Redis dans un conteneur
│   ├── docker-compose.yml # Fichier Docker Compose pour tester plusieurs instances
│── go.mod                 # Fichier de gestion des dépendances Go
│── README.md              # Documentation du projet
